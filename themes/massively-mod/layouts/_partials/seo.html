{{- /* Search-engine and link-preview tags (Facebook, Messenger, WhatsApp, X, …). Description: front-matter `description:`, else for blog posts the same plain-text teaser as the post lists, else a gallery line, else the site description (params.description). Image: the post cover (`image:`), else the first full-size photo of the page (never a .thu crop), else params.ogImage (a file in static/). Served as JPEG, which every link-preview crawler understands. */ -}} {{- $desc := .Description -}} {{- if and (not $desc) .IsPage (eq .Section "blog") -}} {{- $desc = partial "posts/summary.html" . -}} {{- end -}} {{- if and (not $desc) (eq .Layout "single_photos") -}} {{- $desc = printf "%s – fotogalerie Víta Jandáska" .Title -}} {{- end -}} {{- $desc = $desc | default site.Params.description | plainify | strings.TrimSpace -}} {{- $img := false -}} {{- with .Params.image -}}{{- $img = $.Resources.GetMatch . -}}{{- end -}} {{- if not $img -}} {{- range .Resources.ByType "image" -}} {{- if and (not $img) (not (findRE `\.thu\.[^./]+$` .Name)) -}}{{- $img = . -}}{{- end -}} {{- end -}} {{- end -}} {{- $isPost := and .IsPage (eq .Section "blog") -}} {{- $title := cond .IsHome site.Title .Title }} {{- with site.GetPage "/blog" }}{{ with .OutputFormats.Get "rss" }} {{- end }}{{ end }} {{- if $isPost }}{{ with .Date }}{{ if not .IsZero }} {{- end }}{{ end }}{{ range .Params.tags }} {{- end }}{{ end }} {{- with $img }}{{ $og := .Fit "1200x1200 q80 jpg" }} {{- else }}{{ with site.Params.ogImage }} {{- end }}{{ end }}
